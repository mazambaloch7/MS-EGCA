def find_dataset_root():
    candidates = [CONFIG["DATA_ROOT"], "./data", "./dataset", "./millitary-vechiles", "./military-vehicles",
                  "./Millitary_Vehicles", "D:/datasets/millitary-vechiles/", "D:/datasets/military-vehicles/"]
    img_exts = {'.jpg','.jpeg','.png','.bmp','.webp'}
    for cand in candidates:
        p = Path(cand)
        if not p.exists(): continue
        subdirs = [d for d in p.iterdir() if d.is_dir()]
        valid = []
        for sd in subdirs:
            imgs = sum(1 for _ in sd.rglob("*") if _.suffix.lower() in img_exts)
            if imgs >= 5: valid.append(sd)
        if len(valid) >= 2: return str(p)
        for root, dirs, files in os.walk(p):
            cur = Path(root)
            count = 0
            for ss in cur.iterdir():
                if ss.is_dir() and any(f.suffix.lower() in img_exts for f in ss.iterdir() if f.is_file()):
                    count += 1
            if count >= 2: return root
    return None

def scan_dataset(root):
    img_exts = {'.jpg','.jpeg','.png','.bmp','.webp'}
    root = Path(root)
    if not root.exists(): raise FileNotFoundError(root)
    class_dirs = [d for d in root.iterdir() if d.is_dir()]
    valid_class_dirs = []
    for cd in class_dirs:
        imgs = [i for i in cd.rglob("*") if i.suffix.lower() in img_exts]
        if len(imgs)>0: valid_class_dirs.append(cd)
    if len(valid_class_dirs) < 2:
        for cd in class_dirs:
            for ss in cd.iterdir():
                if ss.is_dir():
                    imgs = [i for i in ss.rglob("*") if i.suffix.lower() in img_exts]
                    if len(imgs)>0: valid_class_dirs.append(ss)
    class_names = sorted(list(set([d.name for d in valid_class_dirs])))
    class_to_idx = {n:i for i,n in enumerate(class_names)}
    records, corrupted, duplicates, hashes = [], [], [], {}
    for cd in valid_class_dirs:
        cname = cd.name
        if cname not in class_to_idx:
            for cn in class_names:
                if cn in str(cd): cname = cn; break
        cidx = class_to_idx.get(cname, 0)
        for img_path in cd.rglob("*"):
            if img_path.suffix.lower() not in img_exts: continue
            try:
                with Image.open(img_path) as im: im.verify()
                with Image.open(img_path) as im: w,h = im.size
                fsize = img_path.stat().st_size
                with open(img_path,'rb') as fbin: hh = hashlib.md5(fbin.read()).hexdigest()
                if hh in hashes:
                    duplicates.append({"image_path": str(img_path), "duplicate_of": hashes[hh], "hash": hh})
                else: hashes[hh] = str(img_path)
                status="ok"
            except Exception as e:
                corrupted.append({"image_path": str(img_path), "error": str(e)})
                continue
            records.append({"image_path": str(img_path), "class_name": cname, "class_index": cidx,
                            "width": w, "height": h, "file_size": fsize, "status": status, "hash": hh})
    df = pd.DataFrame(records)
    return df, pd.DataFrame(corrupted), pd.DataFrame(duplicates), class_names, class_to_idx

def get_transforms(img_size=224, train=True):
    if not TORCHVISION_AVAILABLE: return lambda x: torch.zeros(3, img_size, img_size)
    mean=[0.485,0.456,0.406]; std=[0.229,0.224,0.225]
    if train:
        tr=[T.Resize(int(img_size*1.15)), T.RandomResizedCrop(img_size, scale=(0.7,1.0)), T.RandomHorizontalFlip()]
        if CONFIG["USE_RANDAUGMENT"]:
            try: tr.append(T.RandAugment(num_ops=2, magnitude=7))
            except: pass
        tr += [T.ColorJitter(0.2,0.2,0.2,0.05), T.ToTensor(), T.Normalize(mean,std)]
        try: tr.append(T.RandomErasing(p=0.25))
        except: pass
        return T.Compose(tr)
    else:
        return T.Compose([T.Resize(int(img_size*1.14)), T.CenterCrop(img_size), T.ToTensor(), T.Normalize(mean,std)])

train_transform = get_transforms(CONFIG["IMG_SIZE"], True)
val_transform = get_transforms(CONFIG["IMG_SIZE"], False)

class MilitaryVehicleDataset(Dataset):
    def __init__(self, df, transform=None):
        self.df = df.reset_index(drop=True) if len(df)>0 else df
        self.transform = transform
    def __len__(self): return len(self.df)
    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        path = row["image_path"]; label = int(row["class_index"])
        try: img = Image.open(path).convert("RGB")
        except: img = Image.new("RGB", (CONFIG["IMG_SIZE"], CONFIG["IMG_SIZE"]), (0,0,0))
        if self.transform: img = self.transform(img)
        return img, label, path

def make_loaders(train_df, val_df, test_df, batch_size):
    train_ds = MilitaryVehicleDataset(train_df, train_transform)
    val_ds = MilitaryVehicleDataset(val_df, val_transform)
    test_ds = MilitaryVehicleDataset(test_df, val_transform)
    train_loader = DataLoader(train_ds, batch_size=batch_size, shuffle=True, num_workers=CONFIG["NUM_WORKERS"], pin_memory=False)
    val_loader = DataLoader(val_ds, batch_size=batch_size, shuffle=False, num_workers=CONFIG["NUM_WORKERS"], pin_memory=False)
    test_loader = DataLoader(test_ds, batch_size=batch_size, shuffle=False, num_workers=CONFIG["NUM_WORKERS"], pin_memory=False)
    return train_loader, val_loader, test_loader

class MixUpCutMix:
    def __init__(self, mixup_alpha=0.4, cutmix_alpha=1.0, prob=0.5, switch_prob=0.5):
        self.mixup_alpha=mixup_alpha; self.cutmix_alpha=cutmix_alpha; self.prob=prob; self.switch_prob=switch_prob
    def __call__(self, x, y):
        if random.random() > self.prob or not CONFIG["USE_MIXUP"]:
            return x, F.one_hot(y, num_classes=num_classes).float()
        if random.random() < self.switch_prob and CONFIG["USE_CUTMIX"]:
            return self.cutmix(x,y)
        else: return self.mixup(x,y)
    def mixup(self, x, y):
        lam = np.random.beta(self.mixup_alpha, self.mixup_alpha)
        batch_size = x.size(0); index = torch.randperm(batch_size, device=x.device)
        mixed_x = lam*x + (1-lam)*x[index]
        y_one = F.one_hot(y, num_classes=num_classes).float()
        y_mix = lam*y_one + (1-lam)*y_one[index]
        return mixed_x, y_mix
    def cutmix(self, x, y):
        lam = np.random.beta(self.cutmix_alpha, self.cutmix_alpha)
        B,_,H,W = x.size(); index = torch.randperm(B, device=x.device)
        y_one = F.one_hot(y, num_classes=num_classes).float()
        cut_rat = np.sqrt(1.-lam); cut_w = int(W*cut_rat); cut_h = int(H*cut_rat)
        cx = np.random.randint(W); cy = np.random.randint(H)
        bbx1 = np.clip(cx-cut_w//2,0,W); bby1 = np.clip(cy-cut_h//2,0,H)
        bbx2 = np.clip(cx+cut_w//2,0,W); bby2 = np.clip(cy+cut_h//2,0,H)
        x[:, :, bby1:bby2, bbx1:bbx2] = x[index, :, bby1:bby2, bbx1:bbx2]
        lam = 1 - ((bbx2-bbx1)*(bby2-bby1)/(W*H))
        y_mix = lam*y_one + (1-lam)*y_one[index]
        return x, y_mix

mix_augmentor = MixUpCutMix()

def count_parameters(model): return sum(p.numel() for p in model.parameters() if p.requires_grad)
def estimate_model_size_mb(model):
    param_size = sum(p.numel()*p.element_size() for p in model.parameters())
    buffer_size = sum(b.numel()*b.element_size() for b in model.buffers())
    return (param_size+buffer_size)/1024**2

def measure_latency(model, loader, dev, max_batches=8):
    model.eval(); times=[]
    with torch.no_grad():
        count=0
        for imgs,_,_ in loader:
            if count>=max_batches: break
            imgs = imgs.to(dev)
            start=time.time()
            out = model(imgs)
            if isinstance(out, dict): out = out["logits"]
            end=time.time()
            times.append((end-start)/imgs.size(0)*1000)
            count+=1
    return float(np.mean(times)) if times else 0.0

def create_timm_model(model_name, num_classes, pretrained=True, freeze_backbone=False):
    if not TIMM_AVAILABLE: raise RuntimeError("timm not available")
    try:
        model = timm.create_model(model_name, pretrained=pretrained, num_classes=num_classes)
        used_pretrained = pretrained
    except Exception as e:
        print(f"timm create {model_name} failed {e}, trying pretrained=False")
        model = timm.create_model(model_name, pretrained=False, num_classes=num_classes)
        used_pretrained = False
    if freeze_backbone:
        for p in model.parameters(): p.requires_grad=False
        for name, p in model.named_parameters():
            if any(k in name.lower() for k in ["head","classifier","fc"]):
                p.requires_grad=True
    return model, used_pretrained

        seen=set(); backbones_try = [b for b in order if not (b in seen or seen.add(b))]
        backbone=None; feat_dim=None; backbone_name_used=None; pretrained_used=False; fallback_reason=""
        for i, bb_name in enumerate(backbones_try):
            try:
                backbone = timm.create_model(bb_name, pretrained=True, features_only=True)
                feat_dim = backbone.feature_info.channels()[-1]
                backbone_name_used = bb_name
                pretrained_used = True
                fallback_reason = "success" if i==0 else f"fallback from {pref} after {i} tries"
                break
            except Exception as e:
                fallback_reason = str(e)[:120]
                continue
        if backbone is None:
            backbone = timm.create_model("resnet50", pretrained=False, features_only=True)
            feat_dim = backbone.feature_info.channels()[-1]
            backbone_name_used = "resnet50_fallback_no_pretrain"
            pretrained_used = False
            fallback_reason = "all foundation backbones failed"
        self.backbone = backbone
        self.backbone_name = backbone_name_used
        PROPOSED_BACKBONE_USED["name"] = backbone_name_used
        PROPOSED_BACKBONE_USED["pretrained"] = pretrained_used
        BACKBONE_FALLBACK_LOG.append({
            "Requested_Model": pref,
            "Actual_Model_Used": backbone_name_used,
            "Pretrained_Used": pretrained_used,
            "Fallback_Reason": fallback_reason,
            "Stage": "PatchFeatureExtractor.__init__"
        })
        self.proj = nn.Conv2d(feat_dim, embed_dim, 1)
        self.embed_dim = embed_dim
    def forward(self, x):
        feats = self.backbone(x)[-1]
        feats = self.proj(feats)
        B,C,H,W = feats.shape
        tokens = feats.flatten(2).transpose(1,2)
        N = tokens.shape[1]
        max_tokens = CONFIG["MAX_PATCH_TOKENS"]
        if N > max_tokens: tokens = tokens[:, :max_tokens, :]
        return tokens, (H,W)

class KNNGraphBuilder(nn.Module):
    def __init__(self, k=6): super().__init__(); self.k=k
    def forward(self, tokens):
        B,N,C = tokens.shape
        try:
            norm = F.normalize(tokens, dim=-1)
            sim = torch.bmm(norm, norm.transpose(1,2))
        except Exception:
            norm = F.normalize(tokens.cpu(), dim=-1)
            sim = torch.bmm(norm, norm.transpose(1,2)).to(tokens.device)
        k = min(self.k, N-1)
        sim = sim - torch.eye(N, device=sim.device).unsqueeze(0)*1e6
        vals, idx = torch.topk(sim, k=k, dim=-1)
        return idx, vals

class GraphAttentionLayer(nn.Module):
    def __init__(self, dim, heads=4, dropout=0.2):
        super().__init__()
        assert dim % heads == 0
        self.heads=heads; self.dim_head=dim//heads; self.scale=self.dim_head**-0.5
        self.qkv = nn.Linear(dim, dim*3)
        self.proj = nn.Linear(dim, dim)
        self.dropout = nn.Dropout(dropout)
        self.norm = nn.LayerNorm(dim)
    def forward(self, tokens, neighbor_idx):
        B,N,C = tokens.shape; K = neighbor_idx.shape[-1]; h=self.heads; dh=self.dim_head
        residual = tokens
        qkv = self.qkv(tokens).reshape(B,N,3,h,dh).permute(2,0,3,1,4)
        q,k,v = qkv[0], qkv[1], qkv[2]
        neighbor_k_list=[]; neighbor_v_list=[]
        for b in range(B):
            ni = neighbor_idx[b]
            kb = k[b]
            nk = kb[:, ni, :]
            nv = v[b][:, ni, :]
            neighbor_k_list.append(nk); neighbor_v_list.append(nv)
        neighbor_k = torch.stack(neighbor_k_list, dim=0)
        neighbor_v = torch.stack(neighbor_v_list, dim=0)
        q = q.unsqueeze(3)
        attn = (q * neighbor_k).sum(-1) * self.scale
        attn = attn.softmax(dim=-1); attn = self.dropout(attn)
        out = (attn.unsqueeze(-1) * neighbor_v).sum(3)
        out = out.transpose(1,2).reshape(B,N,C)
        out = self.proj(out)
        out = self.norm(out + residual)
        attn_mean = attn.mean(dim=1)
        return out, attn_mean

class MPGR(nn.Module):
    def __init__(self, dim, layers=2, heads=4, dropout=0.2):
        super().__init__()
        self.layers = nn.ModuleList([GraphAttentionLayer(dim, heads, dropout) for _ in range(layers)])
    def forward(self, tokens, neighbor_idx):
        attns=[]; x=tokens
        for layer in self.layers:
            x, attn = layer(x, neighbor_idx); attns.append(attn)
        return x, attns[-1] if attns else None

class DiscriminativePatchAttention(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.score = nn.Sequential(nn.Linear(dim, dim//2), nn.Tanh(), nn.Linear(dim//2, 1))
    def forward(self, tokens):
        scores = self.score(tokens).squeeze(-1)
        attn = torch.softmax(scores, dim=-1)
        pooled = torch.bmm(attn.unsqueeze(1), tokens).squeeze(1)
        return pooled, attn

class MultiBranchTokenFusion(nn.Module):
    """ Multi-branch token fusion: global + graph + attn + max
        True backbone multi-scale available via CONFIG["USE_TRUE_MULTISCALE"]
    """
    def __init__(self, dim, num_classes):
        super().__init__()
        self.fuse = nn.Sequential(
            nn.Linear(dim*4, dim),
            nn.LayerNorm(dim),
            nn.GELU(),
            nn.Dropout(CONFIG["DROPOUT"])
        )
    def forward(self, global_feat, graph_feat, attn_feat, max_feat):
        x = torch.cat([global_feat, graph_feat, attn_feat, max_feat], dim=-1)
        return self.fuse(x)

class PrototypeContrastiveHead(nn.Module):
    def __init__(self, dim, num_classes, temperature=0.07):
        super().__init__()
        self.prototypes = nn.Parameter(torch.randn(num_classes, dim))
        nn.init.xavier_uniform_(self.prototypes)
        self.temperature = nn.Parameter(torch.tensor(temperature))
    def forward(self, x):
        x_norm = F.normalize(x, dim=-1)
        p_norm = F.normalize(self.prototypes, dim=-1)
        logits = x_norm @ p_norm.t() / self.temperature.clamp(min=0.01)
        return logits, x_norm

class MilGraphCalibNet(nn.Module):
    def __init__(self, num_classes, embed_dim=384, use_gnn=True, use_multiscale=True, use_patch_attention=True, use_prototype_head=True):
        super().__init__()
        self.use_gnn=use_gnn; self.use_multiscale=use_multiscale; self.use_patch_attention=use_patch_attention; self.use_prototype_head=use_prototype_head
        self.patch_extractor = PatchFeatureExtractor(embed_dim=embed_dim)
        self.graph_builder = KNNGraphBuilder(k=CONFIG["KNN_K"])
        if use_gnn: self.mpgr = MPGR(embed_dim, layers=CONFIG["GAT_LAYERS"], heads=CONFIG["GAT_HEADS"], dropout=CONFIG["DROPOUT"])
        if use_patch_attention: self.patch_attn = DiscriminativePatchAttention(embed_dim)
        if use_multiscale: self.fusion = MultiBranchTokenFusion(embed_dim, num_classes)
        self.dropout = nn.Dropout(CONFIG["DROPOUT"])
        if use_prototype_head: self.head = PrototypeContrastiveHead(embed_dim, num_classes)
        else: self.classifier = nn.Linear(embed_dim, num_classes)
        self.num_classes = num_classes
    def forward(self, x):
        tokens, hw = self.patch_extractor(x)
        B,N,C = tokens.shape
        graph_attn=None
        if self.use_gnn:
            try:
                neighbor_idx, neighbor_sim = self.graph_builder(tokens)
                tokens_g, graph_attn = self.mpgr(tokens, neighbor_idx)
            except Exception: tokens_g = tokens; graph_attn=None
        else: tokens_g = tokens
        global_feat = tokens_g.mean(dim=1)
        if self.use_patch_attention:
            attn_feat, patch_attn = self.patch_attn(tokens_g)
        else:
            attn_feat = global_feat
            patch_attn = torch.ones(B, N, device=x.device)/N
        graph_feat = tokens_g.mean(dim=1)
        max_feat = tokens_g.max(dim=1).values
        if self.use_multiscale:
            fused = self.fusion(global_feat, graph_feat, attn_feat, max_feat)
        else:
            fused = global_feat
        fused = self.dropout(fused)
        if self.use_prototype_head:
            logits, embeddings = self.head(fused)
        else:
            logits = self.classifier(fused); embeddings = fused
        return {"logits": logits, "features": embeddings, "patch_tokens": tokens_g, "patch_attention": patch_attn, "graph_attention": graph_attn}

# LOSSES
# ============================================================
class SoftTargetCrossEntropy(nn.Module):
    def forward(self, logits, target):
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        log_probs = F.log_softmax(logits, dim=-1)
        loss = -(target * log_probs).sum(dim=-1).mean()
        return loss

class FocalLoss(nn.Module):
    def __init__(self, gamma=2.0): super().__init__(); self.gamma=gamma
    def forward(self, logits, target):
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        ce = -(target * F.log_softmax(logits, dim=-1)).sum(dim=-1)
        pt = torch.exp(-ce)
        loss = ((1-pt)**self.gamma * ce).mean()
        return loss

class BrierLoss(nn.Module):
    def forward(self, logits, target):
        probs = F.softmax(logits, dim=-1)
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        return ((probs - target)**2).mean()

class SupervisedContrastiveLoss(nn.Module):
    def __init__(self, temperature=0.07): super().__init__(); self.temp=temperature
    def forward(self, features, labels):
        if features is None: return torch.tensor(0.0, device=labels.device)
        features = F.normalize(features, dim=1)
        B = features.shape[0]
        if B < 2: return torch.tensor(0.0, device=features.device)
        sim = features @ features.t() / self.temp
        labels = labels.contiguous().view(-1,1)
        mask = torch.eq(labels, labels.T).float().to(features.device)
        logits_mask = torch.ones_like(mask) - torch.eye(B, device=features.device)
        mask = mask * logits_mask
        exp_sim = torch.exp(sim) * logits_mask
        log_prob = sim - torch.log(exp_sim.sum(1, keepdim=True)+1e-9)
        mean_log_prob_pos = (mask * log_prob).sum(1) / (mask.sum(1)+1e-9)
        loss = -mean_log_prob_pos
        loss = loss[mask.sum(1)>0].mean() if (mask.sum(1)>0).any() else torch.tensor(0.0, device=features.device)
        return loss

class CFRLoss(nn.Module):
    def __init__(self, lambda_focal=0.5, lambda_contrastive=0.1, lambda_calibration=0.05, label_smoothing=0.1):
        super().__init__()
        self.lambda_focal=lambda_focal; self.lambda_contrastive=lambda_contrastive; self.lambda_calibration=lambda_calibration
        self.ce = SoftTargetCrossEntropy(); self.focal=FocalLoss(); self.contrastive=SupervisedContrastiveLoss(); self.brier=BrierLoss()
        self.label_smoothing = label_smoothing if CONFIG["USE_LABEL_SMOOTHING"] else 0.0
    def forward(self, outputs, targets):
        if isinstance(outputs, dict): logits = outputs["logits"]; features = outputs.get("features", None)
        else: logits = outputs; features = None
        if targets.ndim == 1 and self.label_smoothing > 0:
            num_classes = logits.size(-1)
            true_dist = torch.zeros_like(logits)
            true_dist.fill_(self.label_smoothing / (num_classes-1))
            true_dist.scatter_(1, targets.unsqueeze(1), 1.0 - self.label_smoothing)
            soft_targets = true_dist
        elif targets.ndim == 1: soft_targets = F.one_hot(targets, num_classes=logits.size(-1)).float()
        else: soft_targets = targets
        ce_loss = self.ce(logits, soft_targets)
        loss = ce_loss
        if self.lambda_focal > 0: loss = loss + self.lambda_focal * self.focal(logits, soft_targets)
        if self.lambda_contrastive > 0 and features is not None and targets.ndim == 1:
            try: loss = loss + self.lambda_contrastive * self.contrastive(features, targets)
            except: pass
        if self.lambda_calibration > 0: loss = loss + self.lambda_calibration * self.brier(logits, soft_targets)
        return loss, {"ce": ce_loss.item()}

# OPTIMIZER / SCHEDULER
# ============================================================
def get_optimizer(model, config):
    backbone_params=[]; head_params=[]
    for n,p in model.named_parameters():
        if not p.requires_grad: continue
        if "backbone" in n or "patch_extractor.backbone" in n: backbone_params.append(p)
        else: head_params.append(p)
    optimizer = torch.optim.AdamW([
        {"params": backbone_params, "lr": config["BACKBONE_LR"] if backbone_params else config["LR"]},
        {"params": head_params, "lr": config["HEAD_LR"]}
    ], weight_decay=config["WEIGHT_DECAY"])
    return optimizer

def get_scheduler(optimizer, epochs, warmup_epochs=2):
    def lr_lambda(epoch):
        if epoch < warmup_epochs: return float(epoch+1)/warmup_epochs
        progress = (epoch - warmup_epochs) / max(1, epochs - warmup_epochs)
        return 0.5 * (1 + math.cos(math.pi * progress))
    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)

def train_one_epoch(model, loader, optimizer, criterion, device, epoch, use_mixup=True):
    model.train(); total_loss=0; correct=0; total=0
    pbar = tqdm(loader, desc=f"Train {epoch}", leave=False) if TQDM_AVAILABLE else loader
    for imgs, labels, _ in pbar:
        imgs=imgs.to(device); labels=labels.to(device)
        optimizer.zero_grad()
        if use_mixup and CONFIG["USE_MIXUP"]:
            try:
                imgs_mix, targets_mix = mix_augmentor(imgs, labels)
                outputs = model(imgs_mix)
                loss, _ = criterion(outputs, targets_mix)
                hard_labels = labels
            except Exception:
                outputs = model(imgs); loss, _ = criterion(outputs, labels); hard_labels = labels
        else:
            outputs = model(imgs); loss, _ = criterion(outputs, labels); hard_labels = labels
        loss.backward(); torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0); optimizer.step()
        total_loss += loss.item() * imgs.size(0)
        logits = outputs["logits"] if isinstance(outputs, dict) else outputs
        preds = logits.argmax(dim=1)
        correct += (preds == hard_labels).sum().item(); total += imgs.size(0)
        if TQDM_AVAILABLE: pbar.set_postfix(loss=loss.item())
    return total_loss/total if total else 0, correct/total if total else 0

def validate_one_epoch(model, loader, criterion, device):
    model.eval(); total_loss=0; correct=0; total=0; all_preds=[]; all_labels=[]
    with torch.no_grad():
        for imgs, labels, _ in loader:
            imgs=imgs.to(device); labels=labels.to(device)
            outputs = model(imgs); loss,_ = criterion(outputs, labels)
            total_loss += loss.item() * imgs.size(0)
            logits = outputs["logits"] if isinstance(outputs, dict) else outputs
            preds = logits.argmax(dim=1)
            correct += (preds==labels).sum().item(); total += imgs.size(0)
            all_preds.extend(preds.cpu().numpy()); all_labels.extend(labels.cpu().numpy())
    acc = correct/total if total else 0
    macro_f1=0
    if SKLEARN_AVAILABLE and len(set(all_labels))>1:
        try: _,_,f1,_ = precision_recall_fscore_support(all_labels, all_preds, average='macro', zero_division=0); macro_f1=float(f1)
        except: pass
    return total_loss/total if total else 0, acc, macro_f1

def fit_model(model_name, model, train_loader, val_loader, config, loss_config, epochs):
    model = model.to(device)
    criterion = CFRLoss(lambda_focal=loss_config.get("lambda_focal", CONFIG["LAMBDA_FOCAL"]),
                        lambda_contrastive=loss_config.get("lambda_contrastive", CONFIG["LAMBDA_CONTRASTIVE"]),
                        lambda_calibration=loss_config.get("lambda_calibration", CONFIG["LAMBDA_CALIBRATION"]))
    optimizer = get_optimizer(model, config)
    scheduler = get_scheduler(optimizer, epochs)
    best_f1=-1; best_state=None; patience_counter=0
    ckpt_path = CKPT_DIR / f"{model_name}_best.pth"
    if config["RESUME_IF_AVAILABLE"] and ckpt_path.exists():
        try:
            ckpt = torch.load(ckpt_path, map_location=device)
            model.load_state_dict(ckpt["model_state"])
            best_f1 = ckpt.get("best_f1", -1)
            print(f"Resumed {model_name} best_f1={best_f1:.4f}")
            val_loss, val_acc, val_f1 = validate_one_epoch(model, val_loader, criterion, device)
            return model, {"skipped": True, "best_f1": best_f1, "train_time_min": 0, "history": None, "resumed": True}
        except Exception as e:
            print(f"Resume failed for {model_name}: {e}")
    history=[]; train_start=time.time()
    for epoch in range(epochs):
        t0=time.time()
        train_loss, train_acc = train_one_epoch(model, train_loader, optimizer, criterion, device, epoch)
        val_loss, val_acc, val_f1 = validate_one_epoch(model, val_loader, criterion, device)
        scheduler.step()
        epoch_time=time.time()-t0
        history.append([epoch, train_loss, val_loss, train_acc, val_acc, val_f1, epoch_time])
        print(f"[{model_name}] Epoch {epoch+1}/{epochs} Train_loss {train_loss:.4f} acc {train_acc:.4f} | Val_loss {val_loss:.4f} acc {val_acc:.4f} f1 {val_f1:.4f}")
        if val_f1 > best_f1:
            best_f1 = val_f1; patience_counter=0
            if config["SAVE_CHECKPOINTS"]:
                torch.save({"model_state": model.state_dict(), "best_f1": best_f1, "epoch": epoch}, ckpt_path)
                best_state = {k:v.cpu() for k,v in model.state_dict().items()}
        else:
            patience_counter+=1
        if patience_counter >= config["PATIENCE"]:
            print(f"Early stopping at epoch {epoch+1}"); break
    train_time_min = (time.time()-train_start)/60
    if best_state is not None: model.load_state_dict(best_state)
    elif ckpt_path.exists():
        try: ckpt = torch.load(ckpt_path, map_location=device); model.load_state_dict(ckpt["model_state"])
        except: pass
    hist_df = pd.DataFrame(history, columns=["epoch","train_loss","val_loss","train_acc","val_acc","val_f1","epoch_time"])
    hist_df.to_csv(LOG_DIR / f"training_log_{model_name}.csv", index=False)
    return model, {"best_f1": best_f1, "train_time_min": train_time_min, "history": hist_df}

def compute_calibration_metrics(y_true, y_probs):
    y_pred = np.argmax(y_probs, axis=1)
    confidences = np.max(y_probs, axis=1)
    accuracies = (y_pred == y_true)
    n_bins=15; bins=np.linspace(0,1,n_bins+1); ece=0.0; mce=0.0
    for i in range(n_bins):
        mask = (confidences > bins[i]) & (confidences <= bins[i+1])
        if np.sum(mask)>0:
            acc_bin = np.mean(accuracies[mask]); conf_bin = np.mean(confidences[mask]); prop = np.mean(mask)
            gap = abs(acc_bin-conf_bin); ece += prop*gap; mce = max(mce, gap)
    n_classes = y_probs.shape[1]
    y_onehot = np.eye(n_classes)[y_true]
    brier = np.mean(np.sum((y_probs - y_onehot)**2, axis=1))
    eps=1e-12; nll = -np.mean(np.log(y_probs[np.arange(len(y_true)), y_true] + eps))
    return float(ece), float(mce), float(brier), float(nll), float(confidences.mean())

def evaluate_model(model, loader, device, class_names):
    model.eval(); all_probs=[]; all_labels=[]; all_paths=[]
    with torch.no_grad():
        for imgs, labels, paths in loader:
            imgs = imgs.to(device)
            out = model(imgs)
            logits = out["logits"] if isinstance(out, dict) else out
            probs = F.softmax(logits, dim=1).cpu().numpy()
            all_probs.append(probs); all_labels.append(labels.numpy()); all_paths.extend(paths)
    if not all_probs: return {}
    y_probs = np.concatenate(all_probs, axis=0); y_true = np.concatenate(all_labels, axis=0)
    y_pred = np.argmax(y_probs, axis=1)
    acc = accuracy_score(y_true, y_pred)
    precision_macro, recall_macro, f1_macro, _ = precision_recall_fscore_support(y_true, y_pred, average='macro', zero_division=0)
    precision_w, recall_w, f1_w, _ = precision_recall_fscore_support(y_true, y_pred, average='weighted', zero_division=0)
    p_c, r_c, f1_c, support = precision_recall_fscore_support(y_true, y_pred, average=None, zero_division=0, labels=list(range(len(class_names))))
    try:
        if len(np.unique(y_true))>1:
            y_onehot = np.eye(y_probs.shape[1])[y_true]
            roc_auc = roc_auc_score(y_onehot, y_probs, average='macro', multi_class='ovr')
            pr_auc = average_precision_score(y_onehot, y_probs, average='macro')
        else: roc_auc=0.0; pr_auc=0.0
    except: roc_auc=0.0; pr_auc=0.0
    ece,mce,brier,nll,mean_conf = compute_calibration_metrics(y_true, y_probs)
    cm = confusion_matrix(y_true, y_pred)
    return {"accuracy": float(acc), "macro_precision": float(precision_macro), "macro_recall": float(recall_macro),
            "macro_f1": float(f1_macro), "weighted_f1": float(f1_w), "roc_auc_ovr": float(roc_auc), "pr_auc": float(pr_auc),
            "ece": ece, "mce": mce, "brier": brier, "nll": nll, "mean_confidence": mean_conf,
            "y_true": y_true, "y_pred": y_pred, "y_probs": y_probs, "paths": all_paths,
            "per_class": {"precision": p_c, "recall": r_c, "f1": f1_c, "support": support},
            "confusion_matrix": cm}

def save_predictions_csv(model_name, eval_res, class_names):
    if not eval_res or "y_true" not in eval_res or len(eval_res["y_true"])==0:
        return False
    y_true = eval_res["y_true"]; y_pred = eval_res["y_pred"]; y_probs = eval_res["y_probs"]; paths = eval_res["paths"]
    true_class = [class_names[t] if t < len(class_names) else str(t) for t in y_true]
    pred_class = [class_names[p] if p < len(class_names) else str(p) for p in y_pred]
    conf = np.max(y_probs, axis=1); correct = (y_true == y_pred)

    pred_df = pd.DataFrame({
        "image_path": paths, "true_label": y_true, "true_class": true_class,
        "pred_label": y_pred, "pred_class": pred_class,
        "confidence": conf, "correct": correct, "probabilities_json": probs_json
    })
    pred_df.to_csv(PRED_DIR / f"{model_name}_predictions.csv", index=False)
    return True

def temperature_scaling(y_probs, y_true):
    logits = np.log(np.clip(y_probs, 1e-12, 1))
    best_ece=1e9; best_t=1.0
    for t in np.linspace(0.5, 3.0, 26):
        scaled = np.exp(logits / t)
        scaled = scaled / scaled.sum(axis=1, keepdims=True)
        ece,_,_,_,_ = compute_calibration_metrics(y_true, scaled)
        if ece < best_ece: best_ece=ece; best_t=t
    return best_t, best_ece

def apply_corruption(img_pil, name):
    if not CV2_AVAILABLE: return img_pil
    img = np.array(img_pil)
    if "gaussian_noise" in name:
        sigma = float(name.split("_")[-1]) * 255
        noise = np.random.normal(0, sigma, img.shape).astype(np.float32)
        noisy = np.clip(img.astype(np.float32) + noise, 0, 255).astype(np.uint8)
        return Image.fromarray(noisy)
    elif "motion_blur" in name:
        k = 3 if "k3" in name else 5
        kernel = np.zeros((k,k)); kernel[int((k-1)/2), :] = np.ones(k); kernel = kernel / k
        blurred = cv2.filter2D(img, -1, kernel)
        return Image.fromarray(blurred)
    elif "brightness" in name:
        factor = float(name.split("_")[-1])
        enh = np.clip(img.astype(np.float32) * factor, 0, 255).astype(np.uint8)
        return Image.fromarray(enh)
    elif "occlusion" in name:
        pct = int(name.split("_")[-1])/100
        h,w = img.shape[:2]; oh, ow = int(h*pct), int(w*pct)
        img2 = img.copy(); img2[0:oh, 0:ow] = 0
        return Image.fromarray(img2)
    elif "salt_pepper" in name:
        prob=0.05; out=img.copy(); rnd=np.random.rand(*img.shape[:2])
        out[rnd < prob/2] = 0; out[rnd > 1-prob/2] = 255
        return Image.fromarray(out)
    elif "jpeg_quality" in name:
        q = int(name.split("_")[-1])
        from io import BytesIO
        buf = BytesIO(); Image.fromarray(img).save(buf, format='JPEG', quality=q); buf.seek(0)
        return Image.open(buf).convert("RGB")
    elif "low_resolution" in name:
        h,w = img.shape[:2]
        small = cv2.resize(img, (w//2, h//2), interpolation=cv2.INTER_AREA)
        up = cv2.resize(small, (w,h), interpolation=cv2.INTER_NEAREST)
        return Image.fromarray(up)
    else: return img_pil

def robust_evaluate(model, test_df, device, corruptions):
    results=[]; base_transform=val_transform; model.eval()
    for corr in corruptions:
        correct=0; total=0; y_true_all=[]; y_pred_all=[]
        for _, row in test_df.iterrows():
            try:
                img = Image.open(row["image_path"]).convert("RGB")
                if corr != "clean": img = apply_corruption(img, corr)
                tensor = base_transform(img).unsqueeze(0).to(device)
                with torch.no_grad():
                    out = model(tensor)
                    logits = out["logits"] if isinstance(out, dict) else out
                    pred = logits.argmax(1).item()
                true = int(row["class_index"])
                y_true_all.append(true); y_pred_all.append(pred)
                if pred == true: correct+=1
                total+=1
            except Exception: continue
        acc = correct/total if total else 0
        if SKLEARN_AVAILABLE and y_true_all:
            _,_,f1,_ = precision_recall_fscore_support(y_true_all, y_pred_all, average='macro', zero_division=0)
        else: f1=0
        results.append({"perturbation": corr, "accuracy": acc, "macro_f1": f1})
    return results

def bootstrap_ci(y_true, y_pred, metric_func, n_boot=500, alpha=0.05):
    n=len(y_true); scores=[]; rng=np.random.default_rng(CONFIG["SEED"])
    for _ in range(n_boot):
        idx = rng.integers(0,n,n)
        try: scores.append(metric_func(y_true[idx], y_pred[idx]))
        except: pass
    if not scores: return np.nan, np.nan
    lo = np.percentile(scores, 100*alpha/2); hi = np.percentile(scores, 100*(1-alpha/2))
    return float(lo), float(hi)

def mcnemar_test(y_true, pred_a, pred_b):
    a_correct = (pred_a == y_true); b_correct = (pred_b == y_true)
    n01 = np.sum((~a_correct) & b_correct); n10 = np.sum(a_correct & (~b_correct))
    try:
        from scipy.stats import chi2
        if n01+n10==0: return 0.0,1.0
        stat = (abs(n01-n10)-1)**2 / (n01+n10)
        p = 1 - chi2.cdf(stat,1)
        return float(stat), float(p)
    except: return np.nan, np.nan

def cohens_h(p1, p2):
    import math
    if np.isnan(p1) or np.isnan(p2): return np.nan
    p1 = np.clip(p1, 1e-6, 1-1e-6); p2 = np.clip(p2, 1e-6, 1-1e-6)
    return 2*math.asin(math.sqrt(p1)) - 2*math.asin(math.sqrt(p2))

def holm_bonferroni(p_values):
    m = len(p_values)
    sorted_idx = np.argsort(p_values)
    adj = np.zeros(m)
    for rank, idx in enumerate(sorted_idx):
        adj[idx] = min((m - rank) * p_values[idx], 1.0)
    sorted_adj = adj[sorted_idx]
    for i in range(1, m):
        if sorted_adj[i] < sorted_adj[i-1]:
            sorted_adj[i] = sorted_adj[i-1]
    adj[sorted_idx] = sorted_adj
    return adj

class GradCAM:
    def __init__(self, model, target_layer):
        self.model=model; self.target_layer=target_layer; self.gradients=None; self.activations=None
        target_layer.register_forward_hook(self.save_activation)
        target_layer.register_full_backward_hook(self.save_gradient)
    def save_activation(self, module, inp, out): self.activations = out.detach()
    def save_gradient(self, module, grad_in, grad_out): self.gradients = grad_out[0].detach()
    def __call__(self, x, class_idx=None):
        self.model.zero_grad()
        out = self.model(x); logits = out["logits"] if isinstance(out, dict) else out
        if class_idx is None: class_idx = logits.argmax(dim=1)
        loss = logits[0, class_idx]; loss.backward(retain_graph=True)
        grads = self.gradients; acts = self.activations
        weights = grads.mean(dim=(2,3), keepdim=True)
        cam = (weights * acts).sum(dim=1, keepdim=True)
        cam = F.relu(cam)
        cam = F.interpolate(cam, size=x.shape[2:], mode='bilinear', align_corners=False)
        cam = cam - cam.min(); cam = cam / (cam.max() + 1e-8)
        return cam.squeeze().cpu().numpy()

def find_last_conv(model):
    for m in reversed(list(model.modules())):
        if isinstance(m, nn.Conv2d): return m
    return None

def extract_features_timm(model_name, loader, device):
    try:
        backbone = timm.create_model(model_name, pretrained=True, num_classes=0, global_pool='avg')
        backbone = backbone.to(device); backbone.eval()
    except Exception as e:
        return None, None, str(e)
    feats=[]; labels=[]
    with torch.no_grad():
        for imgs, y, _ in loader:
            imgs = imgs.to(device)
            f = backbone(imgs)
            feats.append(f.cpu().numpy()); labels.append(y.numpy())
    if not feats: return None, None, "no features"
    X = np.concatenate(feats, axis=0); y = np.concatenate(labels, axis=0)
    return X, y, "ok"

def run_classical_baseline(name, train_loader, val_loader, test_loader, device, class_names):
    feature_model = "convnext_tiny"
    X_train, y_train, msg = extract_features_timm(feature_model, train_loader, device)
    X_test, y_test, msg2 = extract_features_timm(feature_model, test_loader, device)
    if X_train is None: raise RuntimeError(msg)
    if "rf" in name.lower():
        clf = RandomForestClassifier(n_estimators=200, random_state=CONFIG["SEED"], n_jobs=-1)
    else:
        clf = SVC(kernel='rbf', probability=True, random_state=CONFIG["SEED"])
    clf.fit(X_train, y_train)
    y_pred = clf.predict(X_test)
    y_probs = clf.predict_proba(X_test) if hasattr(clf, "predict_proba") else np.eye(len(class_names))[y_pred]
    if y_probs.shape[1] != len(class_names):
        full_probs = np.zeros((len(y_pred), len(class_names)))
        for i, cls in enumerate(clf.classes_):
            if cls < len(class_names):
                full_probs[:, cls] = y_probs[:, i]
        y_probs = full_probs
    acc = accuracy_score(y_test, y_pred)
    precision_macro, recall_macro, f1_macro, _ = precision_recall_fscore_support(y_test, y_pred, average='macro', zero_division=0)
    ece,mce,brier,nll,mean_conf = compute_calibration_metrics(y_test, y_probs)
    return {
        "accuracy": acc, "macro_precision": precision_macro, "macro_recall": recall_macro, "macro_f1": f1_macro,
        "weighted_f1": f1_macro, "roc_auc_ovr": 0.0, "pr_auc": 0.0,
        "ece": ece, "mce": mce, "brier": brier, "nll": nll, "mean_confidence": mean_conf,
        "y_true": y_test, "y_pred": y_pred, "y_probs": y_probs, "paths": [],
        "per_class": {"precision": np.zeros(len(class_names)), "recall": np.zeros(len(class_names)), "f1": np.zeros(len(class_names)), "support": np.zeros(len(class_names))},
        "confusion_matrix": confusion_matrix(y_test, y_pred)
    }, 0.0, 0

def scan_dataset(root):
    img_exts = {'.jpg','.jpeg','.png','.bmp','.webp'}
    root = Path(root)
    if not root.exists(): raise FileNotFoundError(root)
    class_dirs = [d for d in root.iterdir() if d.is_dir()]
    valid_class_dirs = []
    for cd in class_dirs:
        imgs = [i for i in cd.rglob("*") if i.suffix.lower() in img_exts]
        if len(imgs)>0: valid_class_dirs.append(cd)
    if len(valid_class_dirs) < 2:
        for cd in class_dirs:
            for ss in cd.iterdir():
                if ss.is_dir():
                    imgs = [i for i in ss.rglob("*") if i.suffix.lower() in img_exts]
                    if len(imgs)>0: valid_class_dirs.append(ss)
    class_names = sorted(list(set([d.name for d in valid_class_dirs])))
    class_to_idx = {n:i for i,n in enumerate(class_names)}
    records, corrupted, duplicates, hashes = [], [], [], {}
    for cd in valid_class_dirs:
        cname = cd.name
        if cname not in class_to_idx:
            for cn in class_names:
                if cn in str(cd): cname = cn; break
        cidx = class_to_idx.get(cname, 0)
        for img_path in cd.rglob("*"):
            if img_path.suffix.lower() not in img_exts: continue
            try:
                with Image.open(img_path) as im: im.verify()
                with Image.open(img_path) as im: w,h = im.size
                fsize = img_path.stat().st_size
                with open(img_path,'rb') as fbin: hh = hashlib.md5(fbin.read()).hexdigest()
                if hh in hashes:
                    duplicates.append({"image_path": str(img_path), "duplicate_of": hashes[hh], "hash": hh})
                else: hashes[hh] = str(img_path)
                status="ok"
            except Exception as e:
                corrupted.append({"image_path": str(img_path), "error": str(e)})
                continue
            records.append({"image_path": str(img_path), "class_name": cname, "class_index": cidx,
                            "width": w, "height": h, "file_size": fsize, "status": status, "hash": hh})
    df = pd.DataFrame(records)
    return df, pd.DataFrame(corrupted), pd.DataFrame(duplicates), class_names, class_to_idx

def get_transforms(img_size=224, train=True):
    if not TORCHVISION_AVAILABLE: return lambda x: torch.zeros(3, img_size, img_size)
    mean=[0.485,0.456,0.406]; std=[0.229,0.224,0.225]
    if train:
        tr=[T.Resize(int(img_size*1.15)), T.RandomResizedCrop(img_size, scale=(0.7,1.0)), T.RandomHorizontalFlip()]
        if CONFIG["USE_RANDAUGMENT"]:
            try: tr.append(T.RandAugment(num_ops=2, magnitude=7))
            except: pass
        tr += [T.ColorJitter(0.2,0.2,0.2,0.05), T.ToTensor(), T.Normalize(mean,std)]
        try: tr.append(T.RandomErasing(p=0.25))
        except: pass
        return T.Compose(tr)
    else:
        return T.Compose([T.Resize(int(img_size*1.14)), T.CenterCrop(img_size), T.ToTensor(), T.Normalize(mean,std)])

class MixUpCutMix:
    def __init__(self, mixup_alpha=0.4, cutmix_alpha=1.0, prob=0.5, switch_prob=0.5):
        self.mixup_alpha=mixup_alpha; self.cutmix_alpha=cutmix_alpha; self.prob=prob; self.switch_prob=switch_prob
    def __call__(self, x, y):
        if random.random() > self.prob or not CONFIG["USE_MIXUP"]:
            return x, F.one_hot(y, num_classes=num_classes).float()
        if random.random() < self.switch_prob and CONFIG["USE_CUTMIX"]:
            return self.cutmix(x,y)
        else: return self.mixup(x,y)
    def mixup(self, x, y):
        lam = np.random.beta(self.mixup_alpha, self.mixup_alpha)
        batch_size = x.size(0); index = torch.randperm(batch_size, device=x.device)
        mixed_x = lam*x + (1-lam)*x[index]
        y_one = F.one_hot(y, num_classes=num_classes).float()
        y_mix = lam*y_one + (1-lam)*y_one[index]
        return mixed_x, y_mix
    def cutmix(self, x, y):
        lam = np.random.beta(self.cutmix_alpha, self.cutmix_alpha)
        B,_,H,W = x.size(); index = torch.randperm(B, device=x.device)
        y_one = F.one_hot(y, num_classes=num_classes).float()
        cut_rat = np.sqrt(1.-lam); cut_w = int(W*cut_rat); cut_h = int(H*cut_rat)
        cx = np.random.randint(W); cy = np.random.randint(H)
        bbx1 = np.clip(cx-cut_w//2,0,W); bby1 = np.clip(cy-cut_h//2,0,H)
        bbx2 = np.clip(cx+cut_w//2,0,W); bby2 = np.clip(cy+cut_h//2,0,H)
        x[:, :, bby1:bby2, bbx1:bbx2] = x[index, :, bby1:bby2, bbx1:bbx2]
        lam = 1 - ((bbx2-bbx1)*(bby2-bby1)/(W*H))
        y_mix = lam*y_one + (1-lam)*y_one[index]
        return x, y_mix

mix_augmentor = MixUpCutMix()

def count_parameters(model): return sum(p.numel() for p in model.parameters() if p.requires_grad)
def estimate_model_size_mb(model):
    param_size = sum(p.numel()*p.element_size() for p in model.parameters())
    buffer_size = sum(b.numel()*b.element_size() for b in model.buffers())
    return (param_size+buffer_size)/1024**2

def measure_latency(model, loader, dev, max_batches=8):
    model.eval(); times=[]
    with torch.no_grad():
        count=0
        for imgs,_,_ in loader:
            if count>=max_batches: break
            imgs = imgs.to(dev)
            start=time.time()
            out = model(imgs)
            if isinstance(out, dict): out = out["logits"]
            end=time.time()
            times.append((end-start)/imgs.size(0)*1000)
            count+=1
    return float(np.mean(times)) if times else 0.0

def create_timm_model(model_name, num_classes, pretrained=True, freeze_backbone=False):
    if not TIMM_AVAILABLE: raise RuntimeError("timm not available")
    try:
        model = timm.create_model(model_name, pretrained=pretrained, num_classes=num_classes)
        used_pretrained = pretrained
    except Exception as e:
        print(f"timm create {model_name} failed {e}, trying pretrained=False")
        model = timm.create_model(model_name, pretrained=False, num_classes=num_classes)
        used_pretrained = False
    if freeze_backbone:
        for p in model.parameters(): p.requires_grad=False
        for name, p in model.named_parameters():
            if any(k in name.lower() for k in ["head","classifier","fc"]):
                p.requires_grad=True
    return model, used_pretrained

class PatchFeatureExtractor(nn.Module):
    def __init__(self, embed_dim=384):
        super().__init__()
        pref = CONFIG.get("PROPOSED_BACKBONE", "convnextv2_tiny")
        # V3 full fallback chain
        order = [pref,
                 "convnextv2_tiny.fcmae",
                 "convnext_tiny",
                 "vit_small_patch14_dinov2.lvd142m",
                 "vit_base_patch16_224",
                 "resnet50"]
        seen=set(); backbones_try = [b for b in order if not (b in seen or seen.add(b))]
        backbone=None; feat_dim=None; backbone_name_used=None; pretrained_used=False
        fallback_reason=""
        for i, bb_name in enumerate(backbones_try):
            try:
                backbone = timm.create_model(bb_name, pretrained=True, features_only=True)
                feat_dim = backbone.feature_info.channels()[-1]
                backbone_name_used = bb_name
                pretrained_used = True
                fallback_reason = "success" if i==0 else f"fallback from {pref} after {i} tries"
                break
            except Exception as e:
                fallback_reason = str(e)[:120]
                continue
        if backbone is None:
            try:
                backbone = timm.create_model("resnet50", pretrained=False, features_only=True)
                feat_dim = backbone.feature_info.channels()[-1]
                backbone_name_used = "resnet50_fallback_no_pretrain"
                pretrained_used = False
                fallback_reason = "all foundation backbones failed"
            except Exception as e:
                raise RuntimeError("No backbone available")
        self.backbone = backbone
        self.backbone_name = backbone_name_used
        PROPOSED_BACKBONE_USED["name"] = backbone_name_used
        PROPOSED_BACKBONE_USED["pretrained"] = pretrained_used
        BACKBONE_FALLBACK_LOG.append({
            "Requested_Model": pref,
            "Actual_Model_Used": backbone_name_used,
            "Pretrained_Used": pretrained_used,
            "Fallback_Reason": fallback_reason,
            "Stage": "PatchFeatureExtractor.__init__"
        })
        self.proj = nn.Conv2d(feat_dim, embed_dim, 1)
        self.embed_dim = embed_dim
    def forward(self, x):
        feats = self.backbone(x)[-1]
        feats = self.proj(feats)
        B,C,H,W = feats.shape
        tokens = feats.flatten(2).transpose(1,2)
        N = tokens.shape[1]
        max_tokens = CONFIG["MAX_PATCH_TOKENS"]
        if N > max_tokens: tokens = tokens[:, :max_tokens, :]
        return tokens, (H,W)

class KNNGraphBuilder(nn.Module):
    def __init__(self, k=6): super().__init__(); self.k=k
    def forward(self, tokens):
        B,N,C = tokens.shape
        try:
            norm = F.normalize(tokens, dim=-1)
            sim = torch.bmm(norm, norm.transpose(1,2))
        except Exception:
            norm = F.normalize(tokens.cpu(), dim=-1)
            sim = torch.bmm(norm, norm.transpose(1,2)).to(tokens.device)
        k = min(self.k, N-1)
        sim = sim - torch.eye(N, device=sim.device).unsqueeze(0)*1e6
        vals, idx = torch.topk(sim, k=k, dim=-1)
        return idx, vals

class GraphAttentionLayer(nn.Module):
    def __init__(self, dim, heads=4, dropout=0.2):
        super().__init__()
        assert dim % heads == 0
        self.heads=heads; self.dim_head=dim//heads; self.scale=self.dim_head**-0.5
        self.qkv = nn.Linear(dim, dim*3)
        self.proj = nn.Linear(dim, dim)
        self.dropout = nn.Dropout(dropout)
        self.norm = nn.LayerNorm(dim)
    def forward(self, tokens, neighbor_idx):
        B,N,C = tokens.shape; K = neighbor_idx.shape[-1]; h=self.heads; dh=self.dim_head
        residual = tokens
        qkv = self.qkv(tokens).reshape(B,N,3,h,dh).permute(2,0,3,1,4)
        q,k,v = qkv[0], qkv[1], qkv[2]
        neighbor_k_list=[]; neighbor_v_list=[]
        for b in range(B):
            ni = neighbor_idx[b]
            kb = k[b]
            nk = kb[:, ni, :]
            nv = v[b][:, ni, :]
            neighbor_k_list.append(nk); neighbor_v_list.append(nv)
        neighbor_k = torch.stack(neighbor_k_list, dim=0)
        neighbor_v = torch.stack(neighbor_v_list, dim=0)
        q = q.unsqueeze(3)
        attn = (q * neighbor_k).sum(-1) * self.scale
        attn = attn.softmax(dim=-1); attn = self.dropout(attn)
        out = (attn.unsqueeze(-1) * neighbor_v).sum(3)
        out = out.transpose(1,2).reshape(B,N,C)
        out = self.proj(out)
        out = self.norm(out + residual)
        attn_mean = attn.mean(dim=1)
        return out, attn_mean

class MPGR(nn.Module):
    def __init__(self, dim, layers=2, heads=4, dropout=0.2):
        super().__init__()
        self.layers = nn.ModuleList([GraphAttentionLayer(dim, heads, dropout) for _ in range(layers)])
    def forward(self, tokens, neighbor_idx):
        attns=[]; x=tokens
        for layer in self.layers:
            x, attn = layer(x, neighbor_idx); attns.append(attn)
        return x, attns[-1] if attns else None

class DiscriminativePatchAttention(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.score = nn.Sequential(nn.Linear(dim, dim//2), nn.Tanh(), nn.Linear(dim//2, 1))
    def forward(self, tokens):
        scores = self.score(tokens).squeeze(-1)
        attn = torch.softmax(scores, dim=-1)
        pooled = torch.bmm(attn.unsqueeze(1), tokens).squeeze(1)
        return pooled, attn

class MultiBranchTokenFusion(nn.Module):
    """ Multi-branch token fusion: global + graph + attention + max pool
        If USE_TRUE_MULTISCALE=True and backbone provides multi-level features, use those (not implemented in DirectML pilot – logged)
    """
    def __init__(self, dim, num_classes):
        super().__init__()
        self.fuse = nn.Sequential(
            nn.Linear(dim*4, dim),
            nn.LayerNorm(dim),
            nn.GELU(),
            nn.Dropout(CONFIG["DROPOUT"])
        )
        self.use_true_multiscale = CONFIG["USE_TRUE_MULTISCALE"]
    def forward(self, global_feat, graph_feat, attn_feat, max_feat):
        x = torch.cat([global_feat, graph_feat, attn_feat, max_feat], dim=-1)
        return self.fuse(x)

class PrototypeContrastiveHead(nn.Module):
    def __init__(self, dim, num_classes, temperature=0.07):
        super().__init__()
        self.prototypes = nn.Parameter(torch.randn(num_classes, dim))
        nn.init.xavier_uniform_(self.prototypes)
        self.temperature = nn.Parameter(torch.tensor(temperature))
    def forward(self, x):
        x_norm = F.normalize(x, dim=-1)
        p_norm = F.normalize(self.prototypes, dim=-1)
        logits = x_norm @ p_norm.t() / self.temperature.clamp(min=0.01)
        return logits, x_norm

class MilGraphCalibNet(nn.Module):
    def __init__(self, num_classes, embed_dim=384, use_gnn=True, use_multiscale=True, use_patch_attention=True, use_prototype_head=True):
        super().__init__()
        self.use_gnn=use_gnn; self.use_multiscale=use_multiscale; self.use_patch_attention=use_patch_attention; self.use_prototype_head=use_prototype_head
        self.patch_extractor = PatchFeatureExtractor(embed_dim=embed_dim)
        self.graph_builder = KNNGraphBuilder(k=CONFIG["KNN_K"])
        if use_gnn: self.mpgr = MPGR(embed_dim, layers=CONFIG["GAT_LAYERS"], heads=CONFIG["GAT_HEADS"], dropout=CONFIG["DROPOUT"])
        if use_patch_attention: self.patch_attn = DiscriminativePatchAttention(embed_dim)
        if use_multiscale: self.fusion = MultiBranchTokenFusion(embed_dim, num_classes)
        self.dropout = nn.Dropout(CONFIG["DROPOUT"])
        if use_prototype_head: self.head = PrototypeContrastiveHead(embed_dim, num_classes)
        else: self.classifier = nn.Linear(embed_dim, num_classes)
        self.num_classes = num_classes
    def forward(self, x):
        tokens, hw = self.patch_extractor(x)
        B,N,C = tokens.shape
        graph_attn=None
        if self.use_gnn:
            try:
                neighbor_idx, neighbor_sim = self.graph_builder(tokens)
                tokens_g, graph_attn = self.mpgr(tokens, neighbor_idx)
            except Exception: tokens_g = tokens; graph_attn=None
        else: tokens_g = tokens
        global_feat = tokens_g.mean(dim=1)
        if self.use_patch_attention:
            attn_feat, patch_attn = self.patch_attn(tokens_g)
        else:
            attn_feat = global_feat
            patch_attn = torch.ones(B, N, device=x.device)/N
        graph_feat = tokens_g.mean(dim=1)
        max_feat = tokens_g.max(dim=1).values
        if self.use_multiscale:
            fused = self.fusion(global_feat, graph_feat, attn_feat, max_feat)
        else:
            fused = global_feat
        fused = self.dropout(fused)
        if self.use_prototype_head:
            logits, embeddings = self.head(fused)
        else:
            logits = self.classifier(fused); embeddings = fused
        return {"logits": logits, "features": embeddings, "patch_tokens": tokens_g, "patch_attention": patch_attn, "graph_attention": graph_attn}

# LOSSES

class SoftTargetCrossEntropy(nn.Module):
    def forward(self, logits, target):
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        log_probs = F.log_softmax(logits, dim=-1)
        loss = -(target * log_probs).sum(dim=-1).mean()
        return loss

class FocalLoss(nn.Module):
    def __init__(self, gamma=2.0): super().__init__(); self.gamma=gamma
    def forward(self, logits, target):
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        ce = -(target * F.log_softmax(logits, dim=-1)).sum(dim=-1)
        pt = torch.exp(-ce)
        loss = ((1-pt)**self.gamma * ce).mean()
        return loss

class BrierLoss(nn.Module):
    def forward(self, logits, target):
        probs = F.softmax(logits, dim=-1)
        if target.ndim == 1: target = F.one_hot(target, num_classes=logits.size(-1)).float()
        return ((probs - target)**2).mean()

class SupervisedContrastiveLoss(nn.Module):
    def __init__(self, temperature=0.07): super().__init__(); self.temp=temperature
    def forward(self, features, labels):
        if features is None: return torch.tensor(0.0, device=labels.device)
        features = F.normalize(features, dim=1)
        B = features.shape[0]
        if B < 2: return torch.tensor(0.0, device=features.device)
        sim = features @ features.t() / self.temp
        labels = labels.contiguous().view(-1,1)
        mask = torch.eq(labels, labels.T).float().to(features.device)
        logits_mask = torch.ones_like(mask) - torch.eye(B, device=features.device)
        mask = mask * logits_mask
        exp_sim = torch.exp(sim) * logits_mask
        log_prob = sim - torch.log(exp_sim.sum(1, keepdim=True)+1e-9)
        mean_log_prob_pos = (mask * log_prob).sum(1) / (mask.sum(1)+1e-9)
        loss = -mean_log_prob_pos
        loss = loss[mask.sum(1)>0].mean() if (mask.sum(1)>0).any() else torch.tensor(0.0, device=features.device)
        return loss

class CFRLoss(nn.Module):
    def __init__(self, lambda_focal=0.5, lambda_contrastive=0.1, lambda_calibration=0.05, label_smoothing=0.1):
        super().__init__()
        self.lambda_focal=lambda_focal; self.lambda_contrastive=lambda_contrastive; self.lambda_calibration=lambda_calibration
        self.ce = SoftTargetCrossEntropy(); self.focal=FocalLoss(); self.contrastive=SupervisedContrastiveLoss(); self.brier=BrierLoss()
        self.label_smoothing = label_smoothing if CONFIG["USE_LABEL_SMOOTHING"] else 0.0
    def forward(self, outputs, targets):
        if isinstance(outputs, dict): logits = outputs["logits"]; features = outputs.get("features", None)
        else: logits = outputs; features = None
        if targets.ndim == 1 and self.label_smoothing > 0:
            num_classes = logits.size(-1)
            true_dist = torch.zeros_like(logits)
            true_dist.fill_(self.label_smoothing / (num_classes-1))
            true_dist.scatter_(1, targets.unsqueeze(1), 1.0 - self.label_smoothing)
            soft_targets = true_dist
        elif targets.ndim == 1: soft_targets = F.one_hot(targets, num_classes=logits.size(-1)).float()
        else: soft_targets = targets
        ce_loss = self.ce(logits, soft_targets)
        loss = ce_loss
        if self.lambda_focal > 0: loss = loss + self.lambda_focal * self.focal(logits, soft_targets)
        if self.lambda_contrastive > 0 and features is not None and targets.ndim == 1:
            try: loss = loss + self.lambda_contrastive * self.contrastive(features, targets)
            except: pass
        if self.lambda_calibration > 0: loss = loss + self.lambda_calibration * self.brier(logits, soft_targets)
        return loss, {"ce": ce_loss.item()}

# OPTIMIZER / SCHEDULER

def get_optimizer(model, config):
    backbone_params=[]; head_params=[]
    for n,p in model.named_parameters():
        if not p.requires_grad: continue
        if "backbone" in n or "patch_extractor.backbone" in n: backbone_params.append(p)
        else: head_params.append(p)
    optimizer = torch.optim.AdamW([
        {"params": backbone_params, "lr": config["BACKBONE_LR"] if backbone_params else config["LR"]},
        {"params": head_params, "lr": config["HEAD_LR"]}
    ], weight_decay=config["WEIGHT_DECAY"])
    return optimizer

def get_scheduler(optimizer, epochs, warmup_epochs=2):
    def lr_lambda(epoch):
        if epoch < warmup_epochs: return float(epoch+1)/warmup_epochs
        progress = (epoch - warmup_epochs) / max(1, epochs - warmup_epochs)
        return 0.5 * (1 + math.cos(math.pi * progress))
    return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)

def train_one_epoch(model, loader, optimizer, criterion, device, epoch, use_mixup=True):
    model.train(); total_loss=0; correct=0; total=0
    pbar = tqdm(loader, desc=f"Train {epoch}", leave=False) if TQDM_AVAILABLE else loader
    for imgs, labels, _ in pbar:
        imgs=imgs.to(device); labels=labels.to(device)
        optimizer.zero_grad()
        if use_mixup and CONFIG["USE_MIXUP"]:
            try:
                imgs_mix, targets_mix = mix_augmentor(imgs, labels)
                outputs = model(imgs_mix)
                loss, _ = criterion(outputs, targets_mix)
                hard_labels = labels
            except Exception:
                outputs = model(imgs); loss, _ = criterion(outputs, labels); hard_labels = labels
        else:
            outputs = model(imgs); loss, _ = criterion(outputs, labels); hard_labels = labels
        loss.backward(); torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0); optimizer.step()
        total_loss += loss.item() * imgs.size(0)
        logits = outputs["logits"] if isinstance(outputs, dict) else outputs
        preds = logits.argmax(dim=1)
        correct += (preds == hard_labels).sum().item(); total += imgs.size(0)
        if TQDM_AVAILABLE: pbar.set_postfix(loss=loss.item())
    return total_loss/total if total else 0, correct/total if total else 0

def validate_one_epoch(model, loader, criterion, device):
    model.eval(); total_loss=0; correct=0; total=0; all_preds=[]; all_labels=[]
    with torch.no_grad():
        for imgs, labels, _ in loader:
            imgs=imgs.to(device); labels=labels.to(device)
            outputs = model(imgs); loss,_ = criterion(outputs, labels)
            total_loss += loss.item() * imgs.size(0)
            logits = outputs["logits"] if isinstance(outputs, dict) else outputs
            preds = logits.argmax(dim=1)
            correct += (preds==labels).sum().item(); total += imgs.size(0)
            all_preds.extend(preds.cpu().numpy()); all_labels.extend(labels.cpu().numpy())
    acc = correct/total if total else 0
    macro_f1=0
    if SKLEARN_AVAILABLE and len(set(all_labels))>1:
        try: _,_,f1,_ = precision_recall_fscore_support(all_labels, all_preds, average='macro', zero_division=0); macro_f1=float(f1)
        except: pass
    return total_loss/total if total else 0, acc, macro_f1

def fit_model(model_name, model, train_loader, val_loader, config, loss_config, epochs):
    model = model.to(device)
    criterion = CFRLoss(lambda_focal=loss_config.get("lambda_focal", CONFIG["LAMBDA_FOCAL"]),
                        lambda_contrastive=loss_config.get("lambda_contrastive", CONFIG["LAMBDA_CONTRASTIVE"]),
                        lambda_calibration=loss_config.get("lambda_calibration", CONFIG["LAMBDA_CALIBRATION"]))
    optimizer = get_optimizer(model, config)
    scheduler = get_scheduler(optimizer, epochs)
    best_f1=-1; best_state=None; patience_counter=0
    ckpt_path = CKPT_DIR / f"{model_name}_best.pth"
    # V3: resume with evaluation
    if config["RESUME_IF_AVAILABLE"] and ckpt_path.exists():
        try:
            ckpt = torch.load(ckpt_path, map_location=device)
            model.load_state_dict(ckpt["model_state"])
            best_f1 = ckpt.get("best_f1", -1)
            print(f"Resumed {model_name} best_f1={best_f1:.4f}")
            val_loss, val_acc, val_f1 = validate_one_epoch(model, val_loader, criterion, device)
            return model, {"skipped": True, "best_f1": best_f1, "train_time_min": 0, "history": None, "resumed": True}
        except Exception as e:
            print(f"Resume failed for {model_name}: {e}")
    history=[]; train_start=time.time()
    for epoch in range(epochs):
        t0=time.time()
        train_loss, train_acc = train_one_epoch(model, train_loader, optimizer, criterion, device, epoch)
        val_loss, val_acc, val_f1 = validate_one_epoch(model, val_loader, criterion, device)
        scheduler.step()
        epoch_time=time.time()-t0
        history.append([epoch, train_loss, val_loss, train_acc, val_acc, val_f1, epoch_time])
        print(f"[{model_name}] Epoch {epoch+1}/{epochs} Train_loss {train_loss:.4f} acc {train_acc:.4f} | Val_loss {val_loss:.4f} acc {val_acc:.4f} f1 {val_f1:.4f}")
        if val_f1 > best_f1:
            best_f1 = val_f1; patience_counter=0
            if config["SAVE_CHECKPOINTS"]:
                torch.save({"model_state": model.state_dict(), "best_f1": best_f1, "epoch": epoch}, ckpt_path)
                best_state = {k:v.cpu() for k,v in model.state_dict().items()}
        else:
            patience_counter+=1
        if patience_counter >= config["PATIENCE"]:
            print(f"Early stopping at epoch {epoch+1}"); break
    train_time_min = (time.time()-train_start)/60
    if best_state is not None: model.load_state_dict(best_state)
    elif ckpt_path.exists():
        try: ckpt = torch.load(ckpt_path, map_location=device); model.load_state_dict(ckpt["model_state"])
        except: pass
    hist_df = pd.DataFrame(history, columns=["epoch","train_loss","val_loss","train_acc","val_acc","val_f1","epoch_time"])
    hist_df.to_csv(LOG_DIR / f"training_log_{model_name}.csv", index=False)
    return model, {"best_f1": best_f1, "train_time_min": train_time_min, "history": hist_df}

def compute_calibration_metrics(y_true, y_probs):
    y_pred = np.argmax(y_probs, axis=1)
    confidences = np.max(y_probs, axis=1)
    accuracies = (y_pred == y_true)
    n_bins=15; bins=np.linspace(0,1,n_bins+1); ece=0.0; mce=0.0
    for i in range(n_bins):
        mask = (confidences > bins[i]) & (confidences <= bins[i+1])
        if np.sum(mask)>0:
            acc_bin = np.mean(accuracies[mask]); conf_bin = np.mean(confidences[mask]); prop = np.mean(mask)
            gap = abs(acc_bin-conf_bin); ece += prop*gap; mce = max(mce, gap)
    n_classes = y_probs.shape[1]
    y_onehot = np.eye(n_classes)[y_true]
    brier = np.mean(np.sum((y_probs - y_onehot)**2, axis=1))
    eps=1e-12; nll = -np.mean(np.log(y_probs[np.arange(len(y_true)), y_true] + eps))
    return float(ece), float(mce), float(brier), float(nll), float(confidences.mean())

def evaluate_model(model, loader, device, class_names):
    model.eval(); all_probs=[]; all_labels=[]; all_paths=[]
    with torch.no_grad():
        for imgs, labels, paths in loader:
            imgs = imgs.to(device)
            out = model(imgs)
            logits = out["logits"] if isinstance(out, dict) else out
            probs = F.softmax(logits, dim=1).cpu().numpy()
            all_probs.append(probs); all_labels.append(labels.numpy()); all_paths.extend(paths)
    if not all_probs: return {}
    y_probs = np.concatenate(all_probs, axis=0); y_true = np.concatenate(all_labels, axis=0)
    y_pred = np.argmax(y_probs, axis=1)
    acc = accuracy_score(y_true, y_pred)
    precision_macro, recall_macro, f1_macro, _ = precision_recall_fscore_support(y_true, y_pred, average='macro', zero_division=0)
    precision_w, recall_w, f1_w, _ = precision_recall_fscore_support(y_true, y_pred, average='weighted', zero_division=0)
    p_c, r_c, f1_c, support = precision_recall_fscore_support(y_true, y_pred, average=None, zero_division=0, labels=list(range(len(class_names))))
    try:
        if len(np.unique(y_true))>1:
            y_onehot = np.eye(y_probs.shape[1])[y_true]
            roc_auc = roc_auc_score(y_onehot, y_probs, average='macro', multi_class='ovr')
            pr_auc = average_precision_score(y_onehot, y_probs, average='macro')
        else: roc_auc=0.0; pr_auc=0.0
    except: roc_auc=0.0; pr_auc=0.0
    ece,mce,brier,nll,mean_conf = compute_calibration_metrics(y_true, y_probs)
    cm = confusion_matrix(y_true, y_pred)
    return {"accuracy": float(acc), "macro_precision": float(precision_macro), "macro_recall": float(recall_macro),
            "macro_f1": float(f1_macro), "weighted_f1": float(f1_w), "roc_auc_ovr": float(roc_auc), "pr_auc": float(pr_auc),
            "ece": ece, "mce": mce, "brier": brier, "nll": nll, "mean_confidence": mean_conf,
            "y_true": y_true, "y_pred": y_pred, "y_probs": y_probs, "paths": all_paths,
            "per_class": {"precision": p_c, "recall": r_c, "f1": f1_c, "support": support},
            "confusion_matrix": cm}

def temperature_scaling(y_probs, y_true):
    logits = np.log(np.clip(y_probs, 1e-12, 1))
    best_ece=1e9; best_t=1.0
    for t in np.linspace(0.5, 3.0, 26):
        scaled = np.exp(logits / t)
        scaled = scaled / scaled.sum(axis=1, keepdims=True)
        ece,_,_,_,_ = compute_calibration_metrics(y_true, scaled)
        if ece < best_ece: best_ece=ece; best_t=t
    return best_t, best_ece

def apply_corruption(img_pil, name):
    if not CV2_AVAILABLE: return img_pil
    img = np.array(img_pil)
    if "gaussian_noise" in name:
        sigma = float(name.split("_")[-1]) * 255
        noise = np.random.normal(0, sigma, img.shape).astype(np.float32)
        noisy = np.clip(img.astype(np.float32) + noise, 0, 255).astype(np.uint8)
        return Image.fromarray(noisy)
    elif "motion_blur" in name:
        k = 3 if "k3" in name else 5
        kernel = np.zeros((k,k)); kernel[int((k-1)/2), :] = np.ones(k); kernel = kernel / k
        blurred = cv2.filter2D(img, -1, kernel)
        return Image.fromarray(blurred)
    elif "brightness" in name:
        factor = float(name.split("_")[-1])
        enh = np.clip(img.astype(np.float32) * factor, 0, 255).astype(np.uint8)
        return Image.fromarray(enh)
    elif "occlusion" in name:
        pct = int(name.split("_")[-1])/100
        h,w = img.shape[:2]; oh, ow = int(h*pct), int(w*pct)
        img2 = img.copy(); img2[0:oh, 0:ow] = 0
        return Image.fromarray(img2)
    elif "salt_pepper" in name:
        prob=0.05; out=img.copy(); rnd=np.random.rand(*img.shape[:2])
        out[rnd < prob/2] = 0; out[rnd > 1-prob/2] = 255
        return Image.fromarray(out)
    elif "jpeg_quality" in name:
        q = int(name.split("_")[-1])
        from io import BytesIO
        buf = BytesIO(); Image.fromarray(img).save(buf, format='JPEG', quality=q); buf.seek(0)
        return Image.open(buf).convert("RGB")
    elif "low_resolution" in name:
        h,w = img.shape[:2]

def robust_evaluate(model, test_df, device, corruptions):
    results=[]; base_transform=val_transform; model.eval()
    for corr in corruptions:
        correct=0; total=0; y_true_all=[]; y_pred_all=[]
        for _, row in test_df.iterrows():
            try:
                img = Image.open(row["image_path"]).convert("RGB")
                if corr != "clean": img = apply_corruption(img, corr)
                tensor = base_transform(img).unsqueeze(0).to(device)
                with torch.no_grad():
                    out = model(tensor)
                    logits = out["logits"] if isinstance(out, dict) else out
                    pred = logits.argmax(1).item()
                true = int(row["class_index"])
                y_true_all.append(true); y_pred_all.append(pred)
                if pred == true: correct+=1
                total+=1
            except Exception: continue
        acc = correct/total if total else 0
        if SKLEARN_AVAILABLE and y_true_all:
            _,_,f1,_ = precision_recall_fscore_support(y_true_all, y_pred_all, average='macro', zero_division=0)
        else: f1=0
        results.append({"perturbation": corr, "accuracy": acc, "macro_f1": f1})
    return results

def bootstrap_ci(y_true, y_pred, metric_func, n_boot=500, alpha=0.05):
    n=len(y_true); scores=[]; rng=np.random.default_rng(CONFIG["SEED"])
    for _ in range(n_boot):
        idx = rng.integers(0,n,n)
        try: scores.append(metric_func(y_true[idx], y_pred[idx]))
        except: pass
    if not scores: return np.nan, np.nan
    lo = np.percentile(scores, 100*alpha/2); hi = np.percentile(scores, 100*(1-alpha/2))
    return float(lo), float(hi)

def mcnemar_test(y_true, pred_a, pred_b):
    a_correct = (pred_a == y_true); b_correct = (pred_b == y_true)
    n01 = np.sum((~a_correct) & b_correct); n10 = np.sum(a_correct & (~b_correct))
    try:
        from scipy.stats import chi2
        if n01+n10==0: return 0.0,1.0
        stat = (abs(n01-n10)-1)**2 / (n01+n10)
        p = 1 - chi2.cdf(stat,1)
        return float(stat), float(p)
    except: return np.nan, np.nan

def cohens_h(p1, p2):
    # effect size for proportions
    import math
    if np.isnan(p1) or np.isnan(p2): return np.nan
    p1 = np.clip(p1, 1e-6, 1-1e-6); p2 = np.clip(p2, 1e-6, 1-1e-6)
    return 2*math.asin(math.sqrt(p1)) - 2*math.asin(math.sqrt(p2))

def holm_bonferroni(p_values):
    # return holm-adjusted p-values
    m = len(p_values)
    sorted_idx = np.argsort(p_values)
    adj = np.zeros(m)
    for rank, idx in enumerate(sorted_idx):
        adj[idx] = min((m - rank) * p_values[idx], 1.0)
    # enforce monotonicity
    sorted_adj = adj[sorted_idx]
    for i in range(1, m):
        if sorted_adj[i] < sorted_adj[i-1]:
            sorted_adj[i] = sorted_adj[i-1]
    adj[sorted_idx] = sorted_adj
    return adj

class GradCAM:
    def __init__(self, model, target_layer):
        self.model=model; self.target_layer=target_layer; self.gradients=None; self.activations=None
        target_layer.register_forward_hook(self.save_activation)
        target_layer.register_full_backward_hook(self.save_gradient)
    def save_activation(self, module, inp, out): self.activations = out.detach()
    def save_gradient(self, module, grad_in, grad_out): self.gradients = grad_out[0].detach()
    def __call__(self, x, class_idx=None):
        self.model.zero_grad()
        out = self.model(x); logits = out["logits"] if isinstance(out, dict) else out
        if class_idx is None: class_idx = logits.argmax(dim=1)
        loss = logits[0, class_idx]; loss.backward(retain_graph=True)
        grads = self.gradients; acts = self.activations
        weights = grads.mean(dim=(2,3), keepdim=True)
        cam = (weights * acts).sum(dim=1, keepdim=True)
        cam = F.relu(cam)
        cam = F.interpolate(cam, size=x.shape[2:], mode='bilinear', align_corners=False)
        cam = cam - cam.min(); cam = cam / (cam.max() + 1e-8)
        return cam.squeeze().cpu().numpy()

def find_last_conv(model):
    for m in reversed(list(model.modules())):
        if isinstance(m, nn.Conv2d): return m
    return None

def extract_features_timm(model_name, loader, device):
    # load feature backbone (no classifier)
    try:
        backbone = timm.create_model(model_name, pretrained=True, num_classes=0, global_pool='avg')
        backbone = backbone.to(device); backbone.eval()
    except Exception as e:
        return None, None, str(e)
    feats=[]; labels=[]
    with torch.no_grad():
        for imgs, y, _ in loader:
            imgs = imgs.to(device)
            f = backbone(imgs)
            feats.append(f.cpu().numpy()); labels.append(y.numpy())
    if not feats: return None, None, "no features"
    X = np.concatenate(feats, axis=0); y = np.concatenate(labels, axis=0)
    return X, y, "ok"

def run_classical_baseline(name, train_loader, val_loader, test_loader, device, class_names):
    # use convnext_tiny features
    feature_model = "convnext_tiny"
    try:
        X_train, y_train, msg = extract_features_timm(feature_model, train_loader, device)
        X_test, y_test, msg2 = extract_features_timm(feature_model, test_loader, device)
        if X_train is None: raise RuntimeError(msg)
        if "rf" in name.lower():
            clf = RandomForestClassifier(n_estimators=200, random_state=CONFIG["SEED"], n_jobs=-1)
        else:
            clf = SVC(kernel='rbf', probability=True, random_state=CONFIG["SEED"])
        clf.fit(X_train, y_train)
        y_pred = clf.predict(X_test)
        y_probs = clf.predict_proba(X_test) if hasattr(clf, "predict_proba") else np.eye(len(class_names))[y_pred]
        # align prob shape
        if y_probs.shape[1] != len(class_names):
            # pad
            full_probs = np.zeros((len(y_pred), len(class_names)))
            for i, cls in enumerate(clf.classes_):
                if cls < len(class_names):
                    full_probs[:, cls] = y_probs[:, i]
            y_probs = full_probs
        acc = accuracy_score(y_test, y_pred)
        precision_macro, recall_macro, f1_macro, _ = precision_recall_fscore_support(y_test, y_pred, average='macro', zero_division=0)
        ece,mce,brier,nll,mean_conf = compute_calibration_metrics(y_test, y_probs)
        return {
            "accuracy": acc, "macro_precision": precision_macro, "macro_recall": recall_macro, "macro_f1": f1_macro,

    except Exception as e:
        raise e
